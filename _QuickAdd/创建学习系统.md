module.exports = async (params) => {
    const { app, variables } = params;

    // 直接读取前一个 AI Assistant 的输出
    const input = variables.output;

    if (!input) {
        new Notice("没有读取到 AI Assistant 的输出");
        console.log("QuickAdd variables:", variables);
        return;
    }

    let data;

    try {
        // 清理 AI 可能产生的 Markdown 代码块
        let cleaned = String(input).trim();

        cleaned = cleaned
            .replace(/^```json\s*/i, "")
            .replace(/^```\s*/i, "")
            .replace(/\s*```$/i, "")
            .trim();

        data = JSON.parse(cleaned);

    } catch (e) {
        new Notice("AI 输出不是有效的 JSON");
        console.log("AI 原始输出:", input);
        console.error(e);
        return;
    }

    async function createFolder(path) {
        if (!app.vault.getAbstractFileByPath(path)) {
            await app.vault.createFolder(path);
        }
    }

    async function createStructure(basePath, structure) {

        for (const folder of structure.folders || []) {

            const folderPath = basePath
                ? `${basePath}/${folder.name}`
                : folder.name;

            await createFolder(folderPath);

            for (const file of folder.files || []) {

    const fileName =
        typeof file === "string"
            ? file
            : file.name;

    const fileContent =
        typeof file === "string"
            ? `# ${fileName}\n\n`
            : (file.content || `# ${fileName}\n\n`);

    const filePath =
        `${folderPath}/${fileName}.md`;

    if (!app.vault.getAbstractFileByPath(filePath)) {
        await app.vault.create(
            filePath,
            fileContent
        );
    }
}

            await createStructure(folderPath, folder);
        }
    }

    const root = data.name;

    if (!root) {
        new Notice("AI 没有提供根文件夹名称");
        return;
    }

    await createFolder(root);
    await createStructure(root, data);

    new Notice(`创建完成：${root}`);
};
