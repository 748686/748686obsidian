module.exports = async (params) => {
    const { quickAddApi } = params;

    const result = await quickAddApi.inputPrompt(
        "请粘贴一张学习流程图片",
        "",
        {
            multiline: true,
            imagePaste: true
        }
    );

    if (!result) {
        new Notice("已取消");
        return;
    }

    new Notice("图片输入成功！");
    console.log("图片输入结果：", result);
};
